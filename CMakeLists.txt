cmake_minimum_required(VERSION 3.16)
project(GreenLightEngine LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(traffic_engine src/engine/TrafficEngine.cpp)
target_include_directories(traffic_engine PUBLIC src)
if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
  target_compile_options(traffic_engine PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(green_light src/main.cpp)
target_link_libraries(green_light PRIVATE traffic_engine)
include(CTest)
if(BUILD_TESTING)
  add_executable(mlp_parity ml_handoff/test_mlp.cpp)
  add_test(NAME mlp_parity COMMAND mlp_parity
    ${CMAKE_CURRENT_SOURCE_DIR}/ml_handoff/model_weights.txt
    ${CMAKE_CURRENT_SOURCE_DIR}/ml_handoff/test_vectors.csv)
  add_executable(simulator_forecast tests/simulator_forecast.cpp)
  target_link_libraries(simulator_forecast PRIVATE traffic_engine)
  add_test(NAME simulator_forecast COMMAND simulator_forecast
    ${CMAKE_CURRENT_SOURCE_DIR}/ml_handoff/model_weights.txt
    ${CMAKE_CURRENT_SOURCE_DIR}/ml_handoff/test_vectors.csv)
  add_executable(controller_benchmark tests/controller_benchmark.cpp)
  target_link_libraries(controller_benchmark PRIVATE traffic_engine)
  add_test(NAME controller_benchmark COMMAND controller_benchmark)
  add_executable(engine_tests tests/engine_tests.cpp)
  target_link_libraries(engine_tests PRIVATE traffic_engine)
  foreach(case conservation timing deterministic waiting safety reactive_limits reactive_simulation reactive_fairness reactive_multiseed)
    add_test(NAME engine_${case} COMMAND engine_tests ${case})
  endforeach()
endif()
